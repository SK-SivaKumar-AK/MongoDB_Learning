-------------------------------------------------------------------------------------------------------------------------------------------------------------
comparision Operator:
db.employee.find( { "details.position" : { $eq : "Tester" } } )

db.employee.find( { "details.position" : { $ne : "Tester" } } )

db.employee.find( { "details.salary" : { $gt : 50000 } } )

db.employee.find( { "details.salary" : { $gte : 50000 } } )

db.employee.find( { "details.salary" : { $lt : 50000 } } )

db.employee.find( { "details.salary" : { $lte : 50000 } } )

db.employee.find( { "details.salary" : { $in : [ 50000 , 60000 ] } } )




-------------------------------------------------------------------------------------------------------------------------------------------------------------
Logical Operator
db.employee.find( { $and : [ { "details.position" : { $eq : "Tester" } } , { "details.salary" : { $eq : 60000 } } ] } )

db.employee.find( { $or : [ { "details.position" : { $eq : "Tester" } } , { "details.salary" : { $eq : 60000 } } ] } )



-------------------------------------------------------------------------------------------------------------------------------------------------------------
Evalution Operator:
db.employee.find( { name : { $regex : "^J" , $option : "i" } } )

db.employee.find( { $where : function(){ return this.name == "SivaKumar"; } } )

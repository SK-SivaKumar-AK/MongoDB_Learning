=================================================================================================================================================================
Read Data:
db.users.find()-----------------(List all datas)

db.users.find().limit(2)-----------------(Limits)

db.users.find().sort({name:1}).limit(4)----------------(sorting order first small then caps)

db.users.find().skip(1).limit(3)---------------(skip and then limits)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.find( { name : "SivaKumar" } )----------------(filter)

db.users.find( { "address.street" : "Raja" } )----------------(propert value search)

db.users.find( { name : "SivaKumar" } , {age:1 , name:1 , _id:0} )--------------------(find particular column)

---------------------------------------------------------------------------------------------------------------------------------------

db.users.find( { name : { $eq : "Kumar" } } )--------------------(equal to )

db.users.find( { name : { $ne : "Kumar" } } )-------------------(Not equal to)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.find( { age : { $gt : 25 } } )-------------------(greater than)

db.users.find( { age : { $gte : 25 } } )-------------------(greater than equal to)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.find( { age : { $lt : 25 } } )-------------------(less than)

db.users.find( { age : { $lte : 25 } } )-------------------(less than equal to)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.find( { age : { $gt : 25 , $lt : 30 } } )-------------------(greater than and lessthan)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.find( { name : { $in : ["SivaKumar" , "Kumar"] } } )-------------------(in array)

db.users.find( { name : { $nin : ["SivaKumar" , "Kumar"] } } )-------------------(not in array)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.find( { age : { $gt : 25 , $lt : 30 } , name : { $in : ["SivaKumar" , "Kumar"] }  }  )-------------------(Two filters applied)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.find( { $and : [ { age : 25 } , { name:"Kumar" } ] } )-------------------(both consition true data)

db.users.find( { $or : [ { age : 25 } , { name:"Kumar" } ] } )-------------------(any one condition true data)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.find( { name : { $exists : true } } )-------------------(present)

db.users.find( { name : { $exists : false } } )-------------------(not present)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.find( { age : { $not : { $lt : 30 } } } )-------------------(not in lessthan)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.find( { $expr : { $gt : ["$col1" , "$col2"] } } )-------------------(compare 2 column)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.countDocuments( { "address.street" : "Raja" } )-------------------(total count)










=================================================================================================================================================================
Update Data:
db.users.updateOne( { age : 25 } , { $set : { age : 35 } } )-------------------(update)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.updateOne( { age : 25 } , { $inc : { age : 3 } } )-------------------(increament and update)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.updateOne( { age : 25 } , { $rename : { age : "years_ago" } } )-------------------(key name update)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.updateOne( { age : 25 } , { $push : { hobbies : "press" } } )-------------------(add one more array and  update)

db.users.updateOne( { age : 25 } , { $pull : { hobbies : "press" } } )-------------------(add one more array and  update)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.updateOne( { age : 25 } , { $unset : { hobbies : "" } } )-------------------(one property delete)

---------------------------------------------------------------------------------------------------------------------------------------
db.users.updateMany( { address : { $exists : true } } , { $unset : {address : ""} } )-------------------(update many so exisits used)
